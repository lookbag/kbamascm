-- SCM Pro v1 Schema
create table if not exists public.vendors (id text primary key, vendor_code text unique not null, vendor_name text not null, country text, city text, contact_name text, email text, phone text, default_currency text default 'USD', payment_terms text, lead_time_mp integer, remarks text, created_at timestamptz default now(), updated_at timestamptz default now());
create table if not exists public.parts (id text primary key, part_no text unique not null, part_name text not null, category text, in_out text, vendor_code text, material text, price numeric, currency text default 'USD', moq integer, leadtime_mp integer, safety_stock text, unit text default 'Pcs', remarks text, created_at timestamptz default now(), updated_at timestamptz default now());
create table if not exists public.products (id text primary key, part_no text unique not null, part_name text not null, customer text, type text default 'mp', price numeric, currency text default 'USD', remarks text, created_at timestamptz default now(), updated_at timestamptz default now());
create table if not exists public.bom (id text primary key, assy_part_no text, part_no text, lr text default '-', qty integer default 1, remarks text, created_at timestamptz default now(), updated_at timestamptz default now());
create table if not exists public.inventory (id text primary key, part_no text, location text, qty integer default 0, lot_no text, date date, po_number text, eta date, remarks text, created_at timestamptz default now(), updated_at timestamptz default now());
create table if not exists public.forecast (id text primary key, assy_part_no text, customer text, created_at timestamptz default now(), updated_at timestamptz default now());
create table if not exists public.polist (id text primary key, po_number text unique not null, vendor_code text, date date, eta date, total_value numeric, currency text default 'USD', item_count integer, progress integer default 0, status text default 'draft', lines text, remarks text, created_at timestamptz default now(), updated_at timestamptz default now());
create table if not exists public.hqtrade (id text primary key, shipment_no text unique not null, vendor_code text, po_ref text, etd date, eta date, invoice_amt numeric, currency text default 'USD', qty integer, status text default 'open', incoterms text, origin text, dest text, doc_invoice_status text, doc_invoice_no text, doc_invoice_date date, doc_packing_status text, doc_packing_no text, doc_packing_date date, doc_bl_status text, doc_bl_no text, doc_bl_date date, doc_co_status text, doc_co_no text, doc_co_date date, doc_entry_status text, doc_entry_no text, doc_entry_date date, doc_pod_status text, doc_pod_no text, doc_pod_date date, remarks text, created_at timestamptz default now(), updated_at timestamptz default now());
create table if not exists public.landedcost (id text primary key, shipment_no text, vendor text, invoice_no text, invoice_amt numeric, fx_rate numeric, ocean numeric, air numeric, inland_ori numeric, inland_dest numeric, insurance numeric, forwarding numeric, duty_rate numeric, vat_rate numeric, brokerage numeric, other numeric, total_qty integer, weight numeric, grand_total_krw numeric, calc_date date, created_at timestamptz default now());
create table if not exists public.vendorkpi (id text primary key, vendor_code text, date date, po_ref text, shipment_ref text, otd numeric, quality numeric, lt_compliance numeric, responsiveness numeric, overall_score numeric, grade text, notes text, created_at timestamptz default now(), updated_at timestamptz default now());
create table if not exists public.alerts (id text primary key, type text, severity text, title text, desc text, meta text, action text, action_label text, read boolean default false, created_at timestamptz default now());
create table if not exists public.import_history (id text primary key, date text, filename text, rows integer, status text, created_at timestamptz default now());
-- RLS: allow all (adjust for production)
do $$ declare t text; begin for t in select unnest(array['vendors','parts','products','bom','inventory','forecast','polist','hqtrade','landedcost','vendorkpi','alerts','import_history']) loop execute format('alter table public.%I enable row level security', t); execute format('create policy if not exists "allow_all" on public.%I for all using (true) with check (true)', t); end loop; end $$;
